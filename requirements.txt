import streamlit as st
import numpy as np
import pandas as pd
import plotly.graph_objects as go

# -----------------------------
# PAGE CONFIG (DARK + MODERN)
# -----------------------------
st.set_page_config(
    page_title="HRV Dashboard",
    layout="wide",
    initial_sidebar_state="collapsed"
)

st.markdown("""
<style>
body {
    background-color: #0e1117;
    color: #ffffff;
}
</style>
""", unsafe_allow_html=True)

st.title("🫀 HRV Analysis Dashboard")

# -----------------------------
# MOCK DATA (for layout testing)
# -----------------------------
np.random.seed(1)

ecg_signal = np.cumsum(np.random.randn(300))   # fake ECG
rr = np.random.normal(0.8, 0.05, 100)          # fake RR

# -----------------------------
# ECG PLOT (TOP FULL WIDTH)
# -----------------------------
def plot_ecg(signal):
    fig = go.Figure()
    fig.add_trace(go.Scatter(y=signal, mode='lines'))
    fig.update_layout(
        height=250,
        margin=dict(l=10, r=10, t=30, b=10),
        paper_bgcolor="#111827",
        plot_bgcolor="#111827",
        title="ECG Signal"
    )
    return fig

st.plotly_chart(plot_ecg(ecg_signal), use_container_width=True)

# -----------------------------
# 3-COLUMN MAIN LAYOUT
# -----------------------------
col1, col2, col3 = st.columns([1.2, 1, 1])

# -----------------------------
# HRV TABLE (LEFT)
# -----------------------------
with col1:
    st.subheader("📊 HRV Measures")

    df = pd.DataFrame({
        "Metric": ["Mean RR", "SDNN", "RMSSD", "pNN50"],
        "Value": [
            np.mean(rr),
            np.std(rr),
            np.sqrt(np.mean(np.diff(rr)**2)),
            np.random.randint(10, 40)
        ]
    })

    st.dataframe(df, use_container_width=True)

# -----------------------------
# RETURN MAP (MIDDLE)
# -----------------------------
with col2:
    st.subheader("📍 Local Return Map")

    fig = go.Figure()
    fig.add_trace(go.Scatter(
        x=rr[:-1],
        y=rr[1:],
        mode='markers'
    ))

    fig.update_layout(
        height=300,
        paper_bgcolor="#111827",
        plot_bgcolor="#111827"
    )

    st.plotly_chart(fig, use_container_width=True)

# -----------------------------
# SPECTRUM (RIGHT)
# -----------------------------
with col3:
    st.subheader("📈 RR Spectrum")

    f = np.linspace(0, 0.5, len(rr))
    spectrum = np.abs(np.fft.fft(rr))[:len(rr)//2]

    fig2 = go.Figure()
    fig2.add_trace(go.Scatter(y=spectrum, mode='lines'))

    fig2.update_layout(
        height=300,
        paper_bgcolor="#111827",
        plot_bgcolor="#111827"
    )

    st.plotly_chart(fig2, use_container_width=True)
